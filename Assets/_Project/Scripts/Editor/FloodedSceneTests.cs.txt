var s = UnityEditor.SceneManagement.EditorSceneManager.OpenScene(
  "Assets/_Project/Scenes/FloodedChase.unity", UnityEditor.SceneManagement.OpenSceneMode.Single);

int fail = 0;
void Check(string w, bool ok, string d = "") { if (!ok) fail++;
  UnityEngine.Debug.Log($"[X] {(ok ? "PASS" : "FAIL")}  {w}{(d != "" ? "  -- " + d : "")}"); }

// Particles must never be on an opaque lit shader -- that is what made blocks.
int litParticles = 0, nullParticles = 0;
foreach (var pr in UnityEngine.Object.FindObjectsByType<UnityEngine.ParticleSystemRenderer>(UnityEngine.FindObjectsSortMode.None)) {
  var m = pr.sharedMaterial;
  if (m == null || m.shader == null) { if (pr.renderMode != UnityEngine.ParticleSystemRenderMode.None) nullParticles++; continue; }
  var n = m.shader.name;
  if (n == "Universal Render Pipeline/Lit" || n == "Standard" || n.StartsWith("Legacy Shaders/")) litParticles++;
}
Check("no particle on an opaque lit shader (the block leaves)", litParticles == 0, $"{litParticles} bad");
Check("no drawing particle without a material", nullParticles == 0, $"{nullParticles}");

// Sky
var sky = UnityEngine.RenderSettings.skybox;
Check("skybox material assigned", sky != null, sky != null ? sky.name : "NONE");
Check("skybox shader compiles", sky != null && sky.shader != null && !UnityEditor.ShaderUtil.ShaderHasError(sky.shader));
var cam = UnityEngine.Camera.main;
Check("camera clears to skybox", cam != null && cam.clearFlags == UnityEngine.CameraClearFlags.Skybox);

// Ammo
var boxes = UnityEngine.Object.FindObjectsByType<MonsterChase.Player.AmmoPickup>(UnityEngine.FindObjectsSortMode.None).Length;
Check($"ammo boxes in the map ({boxes})", boxes >= 5);
Check("ammo counter on the HUD", UnityEngine.Object.FindFirstObjectByType<MonsterChase.UI.AmmoCounter>() != null);
var gun = UnityEngine.Object.FindFirstObjectByType<MonsterChase.Player.Gun>();
Check($"gun carries a reserve ({(gun != null ? gun.Reserve : 0)})", gun != null && gun.Reserve > 0);

// Still-working essentials
Check("ritual state", UnityEngine.Object.FindFirstObjectByType<MonsterChase.Ritual.RitualState>() != null);
Check("no legacy input module", UnityEngine.Object.FindFirstObjectByType<UnityEngine.EventSystems.StandaloneInputModule>() == null);
Check("terrain present", UnityEngine.Terrain.activeTerrain != null);
var ai = UnityEngine.Object.FindFirstObjectByType<MonsterChase.Monster.MonsterAI>();
Check("monster on navmesh", ai != null && UnityEngine.AI.NavMesh.SamplePosition(ai.transform.position, out _, 5f, UnityEngine.AI.NavMesh.AllAreas));
Check("breath wired", UnityEngine.Object.FindFirstObjectByType<MonsterChase.Systems.PlayerBreath>() != null);

var p = UnityEngine.GameObject.FindGameObjectWithTag("Player");
if (p != null) { UnityEditor.Selection.activeGameObject = p;
  var v = UnityEditor.SceneView.lastActiveSceneView;
  if (v != null) { v.LookAt(p.transform.position + UnityEngine.Vector3.up * 2f, v.rotation, 22f); v.Repaint(); } }

UnityEngine.Debug.Log($"[X] ===== {fail} FAILURES =====  (scene left open: {s.name})");
